#include <stdio.h>
#include <math.h>

int main()
{
    int n, original, temp, digit, digits = 0, sum = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    original = n;
    temp = n;

    while (temp != 0)
    {
        digits++;
        temp = temp / 10;
    }

    temp = n;

    while (temp != 0)
    {
        digit = temp % 10;
        sum = sum + (int)pow(digit, digits);
        temp = temp / 10;
    }

    if (sum == original)
        printf("Armstrong number");
    else
        printf("Not an Armstrong number");

    return 0;
}
